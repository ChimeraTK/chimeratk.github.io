69 void DoocsSpectrum::auto_init() {
70 // check if the macro pulse number source has been set if the spectrum is buffered
71 if(_nBuffers > 1) {
72 if(!_macroPulseNumberSource.isInitialised()) {
73 throw ChimeraTK::logic_error(
74 "D_spectrum '" + _processArray.getName() + "' has numberOfBuffers > 1 but not macro pulse number source.");
75 }
76 }
77
78 D_spectrum::auto_init(); // currently does nothing
79 D_spectrum::read(); // reads persisted values
80 modified = false;
81
82 // apply description/unit metadata after the .conf file has been loaded, so that the values provided here win
83 applyDescriptionUnits(nullptr);
84
85 // send the current value to the device
86 if(this->get_access() == 1 ||
87 (_processArray.isWriteable() && !hasOtherPropertiesToUpdate())) { // property is writeable
88 sendToDevice(false);
89 // set DOOCS time stamp, workaround for DOOCS bug (get() always gives current time stamp if no timestamp is set,
90 // which breaks consistency check in ZeroMQ subscriptions after the 4 minutes timeout)
91 D_spectrum::set_stamp();
92 }
93 }

114 void DoocsSpectrum::write(std::ostream& s) {
115 // DOOCS is normally keeping the location lock until everything is written for that location: all D_spectrum and all
116 // other properties. This can take too long (like seconds), which leads to noticable freezes of the UI. As a
117 // work-around we release the lock here, wait some time and acquire the lock again. Since this happens in a separate
118 // thread (svr_writer), this slow-down should not be of any harm.
119 get_eqfct()->unlock();
120 usleep(1000);
121 get_eqfct()->lock();
122 if(!modified || _processArray.isReadOnly()) {
123 return;
124 }
125 modified = false;
126 D_spectrum::write(s);
127 }

144 void DoocsSpectrum::updateDoocsBuffer(const TransferElementID& transferElementId) {
145 // Note: we already own the location lock by specification of the DoocsUpdater
146
147 // There are only the processArray and the macro pulse number in the consistency
148 // group. The limits are coming asynchronously and not for every macro pulse,
149 // so we just take test latest we have.
150 if(!updateConsistency(transferElementId)) {
151 return;
152 }
153
154 doocs::Timestamp timestamp = correctDoocsTimestamp();
155
156 auto sinceEpoch = timestamp.get_seconds_and_microseconds_since_epoch();
157 auto seconds = sinceEpoch.seconds;
158 auto microseconds = sinceEpoch.microseconds;
159
160 // set macro pulse number, buffer number and time stamp
161 size_t ibuf = 0;
162 if(_macroPulseNumberSource.isInitialised()) {
163 ibuf = _macroPulseNumberSource % _nBuffers;
164 macro_pulse(_macroPulseNumberSource, ibuf);
165 }
166 set_tmstmp(seconds, microseconds, ibuf);
167
168 if(_processArray.dataValidity() != ChimeraTK::DataValidity::ok) {
169 // this will set error code correctly in given buffer _and_ in main property attribute, while
170 // d_error() sets it only in latter
171 this->error(stale_data, ibuf);
172 }
173 else {
174 this->error(no_error, ibuf);
175 }
176
177 // fill the spectrum
178 const std::vector<float>& processVector = _processArray;
179 if(_nBuffers == 1) {
180 // We have to fill the spectrum differently if it is unbuffered, as the internal data structures seem to be
181 // completely different.
182 memcpy(spectrum()->d_spect_array.d_spect_array_val, processVector.data(), processVector.size() * sizeof(float));
183 spectrum()->d_spect_array.d_spect_array_len = processVector.size();
184 }
185 else {
186 fill_spectrum(processVector.data(), processVector.size(), ibuf);
187 }
188
189 // mark property as modified, for (optional) persistence
190 modified = true;
191
192 sendAsync(timestamp);
193 }

197 void DoocsSpectrum::updateParameters() {
198 // Note: we already own the location lock by specification of the DoocsUpdater
199 float start, increment;
200 if(_startAccessor.isInitialised()) {
201 start = _startAccessor;
202 }
203 else {
204 start = this->spec_start();
205 }
206 if(_incrementAccessor.isInitialised()) {
207 increment = _incrementAccessor;
208 }
209 else {
210 increment = this->spec_inc();
211 }
212
213 // WORKAROUND: spectrum_parameter modifies the internal timestamp of
214 // the spectrum, this confuses our code that it thinks it has already sent off the
215 // spectrum with this timestamp, bumps the timestamp to enable DOOCS to check for inconsistencies
216 // which then confuses the DAQ since it sees data for the same MP with different timestamps
217 // https://mcs-gitlab.desy.de/doocs/doocs-core-libraries/serverlib/-/issues/35
218 auto oldTimeStamp = this->get_timestamp();
219 spectrum_parameter(this->spec_time(), start, increment, this->spec_status());
220 this->set_timestamp(oldTimeStamp);
221 }
