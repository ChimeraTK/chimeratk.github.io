191 template<typename SELF, typename UserType>
192 void PropertyBase::sendArrayToDevice(SELF* dfct, OneDRegisterAccessor<UserType>& processArray) {
193 constexpr bool isSpectrum = std::is_base_of<D_spectrum, SELF>::value;
194
195 // always get a fresh reference
196 auto processVector = processArray.data();
197 size_t arraySize = processArray.getNElements();
198 auto doocsLen = static_cast<size_t>(dfct->length());
199 if(doocsLen != arraySize) {
200 std::cout << "Warning: Array length mismatch in property " << this->getEqFct()->name() << "/" << dfct->basename()
201 << ": Property has length " << doocsLen << " but " << arraySize << " expected." << std::endl;
202 arraySize = std::min(arraySize, size_t(doocsLen));
203 }
204 if constexpr(isSpectrum) {
205 // FIXME: find the efficient, memcopying function for float
206 // always get a fresh reference
207 for(size_t i = 0; i < arraySize; ++i) {
208 processVector[i] = dfct->read_spectrum((int)i);
209 }
210 }
211 else {
212 // Brute force implementation with a loop. Works for all data types.
213 for(size_t i = 0; i < arraySize; ++i) {
214 processVector[i] = dfct->value(i);
215 }
216 }
217 auto timestamp = dfct->get_timestamp().to_time_point();
218 processArray.write(VersionNumber(timestamp));
219
220 // Correct property length in case of a mismatch.
221 if(doocsLen != arraySize) {
222 if constexpr(isSpectrum) {
223 dfct->length(arraySize);
224 // FIXME - do we need to restore values like for arrays?
225 // it's more difficult with D_spectrum because of the buffered/unbuffered feature
226 }
227 else {
228 dfct->set_length(arraySize);
229 // restore value from ProcessArray, as it may have been destroyed in set_length().
230 for(size_t i = 0; i < arraySize; ++i) {
231 dfct->set_value(processVector[i], i);
232 }
233 }
234 }
235 }
