80bool StepperMotorRotaryTestFixture::waitForState(std::string reqStateName, unsigned timeoutInSeconds = 10) {
81 unsigned sleepPeriodInMilliSec = 10;
82 unsigned timeoutInSteps = timeoutInSeconds * 1000 / sleepPeriodInMilliSec;
83 unsigned cnt = 0;
84 bool isCorrectState = false;
85 do {
86 std::this_thread::sleep_for(std::chrono::milliseconds(sleepPeriodInMilliSec));
87
88 if(_stepperMotor->getState() == reqStateName) {
89 isCorrectState = true;
90 break;
91 }
92 cnt++;
93 } while(cnt <= timeoutInSteps);
94
95 return isCorrectState;
96}
