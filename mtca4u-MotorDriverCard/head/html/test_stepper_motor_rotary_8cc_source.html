82bool StepperMotorRotaryTestFixture::waitForState(std::string reqStateName, unsigned timeoutInSeconds = 10) {
83 unsigned sleepPeriodInMilliSec = 10;
84 unsigned timeoutInSteps = timeoutInSeconds * 1000 / sleepPeriodInMilliSec;
85 unsigned cnt = 0;
86 bool isCorrectState = false;
87 do {
88 std::this_thread::sleep_for(std::chrono::milliseconds(sleepPeriodInMilliSec));
89
90 if(_stepperMotor->getState() == reqStateName) {
91 isCorrectState = true;
92 break;
93 }
94 cnt++;
95 } while(cnt <= timeoutInSteps);
96
97 return isCorrectState;
98}

240BOOST_AUTO_TEST_CASE(testCalibrateRebasesToZero) {
241 // Pretend the hardware counter is somewhere in the middle of the range. The
242 // homing search must find the switch within the search offset from the
243 // origin, so move the dummy back onto the switch.
244 _motorControlerDummy->setActualPosition(1000);
245 _motorControlerDummy->setTargetPosition(1000);
246 _motorControlerDummy->simulateBlockedMotor(false);
247
248 BOOST_CHECK_EQUAL(_stepperMotor->isCalibrated(), false);
249 BOOST_CHECK(_stepperMotor->getCalibrationMode() == CalibrationMode::NONE);
250
251 performCalibrationForTest();
252
253 BOOST_CHECK_EQUAL(getCalibrationFailed(), false);
254 BOOST_CHECK_EQUAL(_stepperMotor->isCalibrated(), true);
255 BOOST_CHECK(_stepperMotor->getCalibrationMode() == CalibrationMode::FULL);
256 BOOST_CHECK(_stepperMotor->getError() == Error::NO_ERROR);
257
258 // The counter has been re-based to the home switch
259 BOOST_CHECK_EQUAL(_stepperMotor->getCurrentPositionInSteps(), 0);
260 BOOST_CHECK_EQUAL(_stepperMotor->getPositiveEndReferenceInSteps(), 0);
261 BOOST_CHECK_EQUAL(_stepperMotor->getNegativeEndReferenceInSteps(), 0);
262
263 // The firmware registers hold the same (counter independent) values
264 BOOST_CHECK_NE(_motorControlerDummy->getCalibrationTime(), 0U);
265 BOOST_CHECK_EQUAL(_motorControlerDummy->getPositiveReferenceSwitchCalibration(), 0);
266 BOOST_CHECK_EQUAL(_motorControlerDummy->getNegativeReferenceSwitchCalibration(), 0);
267
268 // Moving away from the home switch works, and moving back towards it too.
269 // After the re-base the dummy hardware counter still sits on the home switch
270 // (its hardware position was not moved), so the motor can only leave the
271 // switch in the negative direction, which is what a homed stage does anyway.
272 BOOST_CHECK(_stepperMotor->setTargetPositionInSteps(-300) == ExitStatus::SUCCESS);
273 BOOST_CHECK_NO_THROW(_stepperMotor->start());
274 BOOST_CHECK(waitForState("moving", 2));
275 _motorControlerDummy->moveTowardsTarget(1);
276 BOOST_CHECK(waitForCalibrationFinished());
277 BOOST_CHECK_EQUAL(_stepperMotor->getCurrentPositionInSteps(), -300);
278
279 BOOST_CHECK(_stepperMotor->setTargetPositionInSteps(0) == ExitStatus::SUCCESS);
280 BOOST_CHECK_NO_THROW(_stepperMotor->start());
281 BOOST_CHECK(waitForState("moving", 2));
282 _motorControlerDummy->moveTowardsTarget(1);
283 BOOST_CHECK(waitForCalibrationFinished());
284 BOOST_CHECK_EQUAL(_stepperMotor->getCurrentPositionInSteps(), 0);
285}
