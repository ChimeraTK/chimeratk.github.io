20 ReferenceStateMachine::ReferenceStateMachine(ReferenceStepperMotor& stepperMotorWithReference)
21 : BasicStepperMotor::StateMachine(stepperMotorWithReference), _calibrating("calibrating"),
22 _motor(stepperMotorWithReference), _stopAction(false), _moveInterrupted(false) {
23 _idle.setTransition(calibEvent, &_calibrating, [this]() { actionStartCalib(); }, [this]() { actionEndCallback(); });
24
25 _calibrating.setTransition(
26 BasicStepperMotor::StateMachine::stopEvent, &_idle, std::bind(&ReferenceStateMachine::actionStop, this));
27 _calibrating.setTransition(BasicStepperMotor::StateMachine::emergencyStopEvent, &_error, [this] {
28 _stopAction.exchange(true);
29 _moveInterrupted.exchange(true);
30 actionEmergencyStop();
31 });
32 _calibrating.setTransition(BasicStepperMotor::StateMachine::errorEvent, &_error, [] {});
33 }

40 void ReferenceStateMachine::invalidateCalibration(bool reportCalibrationError) {
41 try {
42 _motor._motorController->setCalibrationTime(0);

49 if(reportCalibrationError) {
50 _motor._errorMode.store(Error::CALIBRATION_ERROR);

87 void ReferenceStateMachine::moveToEndSwitch(Sign sign) {
88 {
89 boost::lock_guard<boost::mutex&> lck(_motor._mutex);
90 _motor._motorController->setTargetPosition(
91 _motor._motorController->getActualPosition() + static_cast<int>(sign) * getOffset());
92 }
93 while(_motor._motorController->isMotorMoving()) {
94 std::this_thread::sleep_for(std::chrono::milliseconds(wakeupPeriodInMilliseconds));
95 }
96 }
