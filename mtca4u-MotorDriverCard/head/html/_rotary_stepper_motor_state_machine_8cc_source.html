19 void RotaryStepperMotorStateMachine::performCalibration() {
20 // This function runs in a detached thread (see ReferenceStateMachine::actionStartCalib)
21 // and must not let an exception escape: that would terminate the process
22 // and leave _asyncActionActive stuck, locking the machine in "calibrating".
23 // The guard guarantees the flag is cleared on every exit path.
24 auto asyncActionGuard = cppext::finally([&] { _asyncActionActive.store(false); });
25 std::cerr << "rotary calibration: starting home search" << std::endl;
26 _motor._calibrationFailed.exchange(false);
27 _stopAction.exchange(false);
28 _moveInterrupted.exchange(false);
29 // Shared failure handling. Safe to run even if the communication link is
30 // down: the cleanup I/O is allowed to fail.
31 bool searchTargetCleared = false;
32 auto clearSearchTarget = [&] {
33 searchTargetCleared = false;
34 _motor.resetMotorControllerPositions(0);
35 searchTargetCleared = _motor._motorController->getTargetPosition() == 0;
36 };
37 auto failCalibration = [&] {
38 std::lock_guard<boost::mutex> lock(_motor._mutex);
39 try {
40 clearSearchTarget();
41 }
42 catch(const std::exception& e) {
43 std::cerr << "Could not reset rotary homing target: " << e.what() << std::endl;
44 }
45 invalidateCalibration();
46 try {
47 _motor._motorController->setPositiveReferenceSwitchCalibration(0);
48 _motor._motorController->setNegativeReferenceSwitchCalibration(0);
49 }
50 catch(const std::exception& e) {
51 std::cerr << "RotaryStepperMotorStateMachine::performCalibration(): additionally failed to reset the "
52 << "calibration data: " << e.what() << std::endl;
53 }
54 _motor._calibPositiveEndSwitchInSteps.exchange(0);
55 _motor._calibNegativeEndSwitchInSteps.exchange(0);
56 };
57 try {
58 // Establish a fixed origin for the home search: re-base the hardware
59 // counter (actual and target) to 0. Homing then always starts from a
60 // known position, and a stale target left over from an interrupted move
61 // cannot abort the search with a "position mismatch" either.
62 {
63 boost::lock_guard<boost::mutex&> lck(_motor._mutex);
64 _motor.resetMotorControllerPositions(0);
65 }
66 bool stopDisabled = false;
67 int homePosition = 0;
68 {
69 auto restoreStop = cppext::finally([&] {
70 if(!searchTargetCleared) {
71 std::cerr << "Leaving rotary home stop enabled because the search target could not be cleared" << std::endl;
72 return;
73 }
74 try {
75 _motor._motorController->setPositiveReferenceSwitchEnabled(false);
76 stopDisabled = true;
77 }
78 catch(const std::exception& e) {
79 std::cerr << "Could not disable rotary home stop: " << e.what() << std::endl;
80 }
81 });
82 try {
83 _motor._motorController->setPositiveReferenceSwitchEnabled(true);
84 findEndSwitch(Sign::POSITIVE);
85 homePosition = _motor.getCurrentPositionInSteps();
86 std::unique_lock<boost::mutex> lock(_motor._mutex);
87 if(_moveInterrupted.load() || _stopAction.load()) {
88 lock.unlock();
89 failCalibration();
90 return;
91 }
92 // Clear the search target before releasing the hardware stop.
93 clearSearchTarget();
94 }
95 catch(const std::exception& e) {
96 std::cerr << "Rotary home search failed: " << e.what() << std::endl;
97 failCalibration();
98 return;
99 }
100 }
101
102 std::unique_lock<boost::mutex> lock(_motor._mutex);
103 if(!stopDisabled || _moveInterrupted.load() || _stopAction.load()) {
104 lock.unlock();
105 failCalibration();
106 }
107 else {
108 // The counter was re-based at the home switch. Persist that origin
109 // so the saved calibration is independent of the search offset.
110 std::cerr << "rotary calibration: home switch found at position " << homePosition << " steps" << std::endl;
111 constexpr int calibPositiveEndSwitchInSteps = 0;
112 constexpr int calibNegativeEndSwitchInSteps = 0;
113
114 _motor._motorController->setPositiveReferenceSwitchCalibration(calibPositiveEndSwitchInSteps);
115 _motor._motorController->setNegativeReferenceSwitchCalibration(calibNegativeEndSwitchInSteps);
116 _motor._calibPositiveEndSwitchInSteps.exchange(calibPositiveEndSwitchInSteps);
117 _motor._calibNegativeEndSwitchInSteps.exchange(calibNegativeEndSwitchInSteps);
118 // Written last, so that a restart never sees a valid CAL_TIME together
119 // with half-updated end switch registers (CAL_TIME has no write-back
120 // validity flag in the firmware map, the end switch registers do).
121 _motor._motorController->setCalibrationTime(time(nullptr));
122
123 _motor._calibrationMode.exchange(CalibrationMode::FULL);
124 }
125 }
126 catch(ChimeraTK::runtime_error& e) {
127 std::cerr << "RotaryStepperMotorStateMachine::performCalibration(): homing failed: " << e.what() << std::endl;
128 failCalibration();
129 }
130 catch(const std::exception& e) {
131 std::cerr << "RotaryStepperMotorStateMachine::performCalibration(): unexpected exception during homing: "
132 << e.what() << std::endl;
133 failCalibration();
134 }
135 }

137 void RotaryStepperMotorStateMachine::findEndSwitch(Sign sign) {
138 auto startTime = std::chrono::steady_clock::now();
139 constexpr int HOMING_TIMEOUT_MS = 60000;
140 int loopIterations = 0;
141 int lastPosition = _motor.getCurrentPositionInSteps();
142 while(!_motor.isEndSwitchActive(sign)) {
143 ++loopIterations;
144 if(_stopAction.load() || _moveInterrupted.load()) {
145 std::cerr << "findEndSwitch: stopped by user after " << loopIterations << " iterations" << std::endl;
146 return;
147 }
148 auto now = std::chrono::steady_clock::now();
149 if(std::chrono::duration_cast<std::chrono::milliseconds>(now - startTime).count() > HOMING_TIMEOUT_MS) {
150 std::cerr << "findEndSwitch: timeout after " << loopIterations << " iterations, end switch not found"
151 << std::endl;
152 _moveInterrupted.exchange(true);
153 return;
154 }
155 if(_motor.getTargetPositionInSteps() != _motor.getCurrentPositionInSteps() &&
156 !_motor.isPositiveReferenceActive() && !_motor.isNegativeReferenceActive()) {
157 std::cerr << "findEndSwitch: position mismatch (motor did not move) after " << loopIterations << " iterations"
158 << std::endl;
159 _moveInterrupted.exchange(true);
160 return;
161 }
162 moveToEndSwitch(sign);
163 // moveToEndSwitch() blocks while the motor is moving. A motor that cannot
164 // move at all (e.g. blocked, no current) reports "not moving" instantly, so
165 // without this guard the loop would spin at full speed until the 60 s
166 // timeout. Detect a stalled motor instead.
167 int position = _motor.getCurrentPositionInSteps();
168 if(position == lastPosition) {
169 std::cerr << "findEndSwitch: motor stalled at position " << position << " after " << loopIterations
170 << " iterations" << std::endl;
171 _moveInterrupted.exchange(true);
172 return;
173 }
174 lastPosition = position;
175 }
176 std::cerr << "findEndSwitch: end switch active after " << loopIterations << " iterations" << std::endl;
177 }
