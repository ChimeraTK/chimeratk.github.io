32template<typename Predicate>
33static bool pollUntil(Predicate predicate, std::chrono::milliseconds timeout = std::chrono::seconds(5)) {
34 using clock = std::chrono::steady_clock;
35 const auto start = clock::now();
36 while(clock::now() - start < timeout) {
37 if(predicate()) {
38 return true;
39 }
40 std::this_thread::sleep_for(std::chrono::milliseconds(20));
41 }
42 return false;
43}

50class EventCollector : public Tango::CallBack {
51 public:
52 void push_event(Tango::EventData* ed) override {
53 std::lock_guard<std::mutex> lg(_mutex);
54 if(ed->err) {
55 return;
56 }
57 if(ed->attr_value != nullptr) {
58 _changeEvents.emplace_back(*ed->attr_value);
59 }
60 }
61
62 void push_event(Tango::DataReadyEventData* dre) override {
63 std::lock_guard<std::mutex> lg(_mutex);
64 if(dre->err) {
65 return;
66 }
67 _dataReadyCounters.push_back(dre->ctr);
68 }
69
73 template<typename Predicate>
74 bool hasChangeEvent(Predicate predicate) {
75 std::lock_guard<std::mutex> lg(_mutex);
76 return std::any_of(_changeEvents.begin(), _changeEvents.end(),
77 [&predicate](Tango::DeviceAttribute& attr) { return predicate(attr); });
78 }
79
83 template<typename Predicate>
84 std::optional<Tango::DeviceAttribute> lastChangeEvent(Predicate predicate) {
85 std::lock_guard<std::mutex> lg(_mutex);
86 for(auto& event : std::views::reverse(_changeEvents)) {
87 if(predicate(event)) {
88 return event;
89 }
90 }
91 return std::nullopt;
92 }
93
97 int lastDataReadyCounter() {
98 std::lock_guard<std::mutex> lg(_mutex);
99 assert(!_dataReadyCounters.empty());
100 return _dataReadyCounters.back();
101 }
102
106 [[nodiscard]] size_t dataReadyEventCount() {
107 std::lock_guard<std::mutex> lg(_mutex);
108 return _dataReadyCounters.size();
109 }
110
111 private:
112 std::mutex _mutex;
113 std::vector<Tango::DeviceAttribute> _changeEvents;
114 std::vector<int> _dataReadyCounters;
115};

127template<typename WriteFunction, typename Predicate>
128static bool writeOnceAndRunMainLoopUntil(WriteFunction write, Predicate condition) {
129 write();
130 const auto deadline = std::chrono::steady_clock::now() + std::chrono::seconds(5);
131 do {
132 ExtendedReferenceTestApplication::runMainLoopOnce();
133 if(pollUntil(condition, std::chrono::milliseconds(100))) {
134 return true;
135 }
136 } while(std::chrono::steady_clock::now() < deadline);
137
138 return condition();
139}

146static std::chrono::system_clock::time_point eventTimeStamp(Tango::DeviceAttribute& attribute) {
147 const auto date = attribute.get_date();
148 return std::chrono::system_clock::time_point(std::chrono::seconds(date.tv_sec) +
149 std::chrono::microseconds(date.tv_usec) + std::chrono::nanoseconds(date.tv_nsec));
150}

154BOOST_AUTO_TEST_CASE(testChangeEventCarriesWrittenValue) {
155 auto [tf, app, proxy] = TangoTestFixtureImpl::getContents();
156
157 EventCollector collector;
158 const auto eventId = proxy.subscribe_event("INT_FROM_DEVICE_SCALAR", Tango::CHANGE_EVENT, &collector);
159
160 // The mapper is driven by the source location, and the nested variable name reaches the event machinery: this is
161 // proven by the CHANGE event below carrying the value written to the mapped process variable. The attribute name is
162 // not read back, because Tango sets DeviceAttribute::name to the requested name, so such a read cannot fail.
163 app.dataValidity = ChimeraTK::DataValidity::ok;
164
165 const auto hasWrittenValue = [&collector] {
166 return collector.hasChangeEvent([](Tango::DeviceAttribute& attr) {
167 Tango::DevLong value = 0;
168 attr >> value;
169 return value == 42;
170 });
171 };
172 BOOST_REQUIRE(
173 writeOnceAndRunMainLoopUntil([&tf] { tf.write("INT_TO_DEVICE_SCALAR", Tango::DevLong(42)); }, hasWrittenValue));
174
175 auto event = collector.lastChangeEvent([](Tango::DeviceAttribute& attr) {
176 Tango::DevLong value = 0;
177 attr >> value;
178 return value == 42;
179 });
180 BOOST_REQUIRE(event.has_value());
181 Tango::DevLong value = 0;
182 (*event) >> value;
183 BOOST_TEST(value == 42);
184
185 proxy.unsubscribe_event(eventId);
186}

299BOOST_AUTO_TEST_CASE(testVoidChangeEventCarriesReadoutTimeStamp) {
300 auto [tf, app, proxy] = TangoTestFixtureImpl::getContents();
301
302 EventCollector collector;
303 const auto eventId = proxy.subscribe_event("VOID_FROM_DEVICE_SCALAR_CHANGE", Tango::CHANGE_EVENT, &collector);
304
305 app.dataValidity = ChimeraTK::DataValidity::ok;
306 const auto beforeWrite = std::chrono::system_clock::now();
307
308 // The subscription itself already produces a change event carrying the value the attribute held before the write, so
309 // the event of the write is the one whose time stamp does not lie before the write.
310 const auto producedByWrite = [&beforeWrite](
311 Tango::DeviceAttribute& attr) { return eventTimeStamp(attr) >= beforeWrite; };
312 BOOST_REQUIRE(writeOnceAndRunMainLoopUntil([&tf] { tf.write("VOID_TO_DEVICE_SCALAR", Tango::DevBoolean(true)); },
313 [&collector, &producedByWrite] { return collector.hasChangeEvent(producedByWrite); }));
314 const auto eventArrived = std::chrono::system_clock::now();
315
316 auto event = collector.lastChangeEvent(producedByWrite);
317 BOOST_REQUIRE(event.has_value());
318
319 // A Void register carries no value, so the attribute reports the constant true of its read path.
320 Tango::DevBoolean value = false;
321 (*event) >> value;
322 BOOST_TEST(value == true);
323 BOOST_TEST((*event).get_quality() == Tango::AttrQuality::ATTR_VALID);
324
325 // The event carries the readout time stamp of the process variable, not the reception time of the event: the
326 // predicate above already places it after the write, and it cannot lie after the arrival either.
327 const auto firstTimeStamp = eventTimeStamp(*event);
328 BOOST_TEST(firstTimeStamp <= eventArrived);
329
330 // A second write is a new version of the process variable, so its event carries a later time stamp.
331 const auto hasLaterEvent = [&collector, firstTimeStamp] {
332 return collector
333 .lastChangeEvent(
334 [&firstTimeStamp](Tango::DeviceAttribute& attr) { return eventTimeStamp(attr) > firstTimeStamp; })
335 .has_value();
336 };
337 BOOST_REQUIRE(writeOnceAndRunMainLoopUntil(
338 [&tf] { tf.write("VOID_TO_DEVICE_SCALAR", Tango::DevBoolean(true)); }, hasLaterEvent));
339
340 proxy.unsubscribe_event(eventId);
341}
