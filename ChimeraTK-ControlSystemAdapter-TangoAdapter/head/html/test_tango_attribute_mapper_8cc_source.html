74static void readMapperFor(
75 const std::string& caseName, const std::string& attributesXml, TangoAdapter::AttributeMapper& mapper) {
76 // The process variable manager is the running server's control-system manager, since the application only
77 // exposes a DevicePVManager.
78 mapper.setCSPVManager(TangoAdapter::TangoAdapter::getInstance().getCsPvManager());
79 withMapperFileInCaseDirectory(
80 caseName, std::string(className), std::string(deviceName), attributesXml, [&mapper] { mapper.readMapperFile(); });
81}

86static std::list<TangoAdapter::AttributeProperty> parseAttributesOf(
87 const std::string& caseName, const std::string& attributesXml) {
88 TangoAdapter::AttributeMapper mapper;
89 readMapperFor(caseName, attributesXml, mapper);
90
91 std::string name = std::string(className);
92 const auto deviceClass = mapper.getClass(name);
93 BOOST_REQUIRE(deviceClass != nullptr);
94 return deviceClass->attributes;
95}

100static void expectMapperRejects(const std::string& caseName, const std::string& attributesXml) {
101 TangoAdapter::AttributeMapper mapper;
102 BOOST_CHECK_THROW(readMapperFor(caseName, attributesXml, mapper), ChimeraTK::logic_error);
103}

110BOOST_AUTO_TEST_CASE(testMapperParsesEventing) {
111 const std::vector<std::pair<std::string, TangoAdapter::EventingMode>> cases{
112 {"dataReady", TangoAdapter::EventingMode::DataReady},
113 {"change", TangoAdapter::EventingMode::Change},
114 {"off", TangoAdapter::EventingMode::Off},
115 {"", TangoAdapter::EventingMode::Off},
116 };
117
118 for(const auto& [text, expected] : cases) {
119 const auto attributes = parseAttributesOf(text.empty() ? "absent" : text,
120 attributeWithEventing("INT_FROM_DEVICE_SCALAR", "INT/FROM_DEVICE_SCALAR", text));
121
122 BOOST_REQUIRE_EQUAL(attributes.size(), 1U);
123 BOOST_TEST(attributes.front().eventing == expected);
124 }
125}

145BOOST_AUTO_TEST_CASE(testMapperMapsScalarAttributeWithEventing) {
146 const auto attributes = parseAttributesOf("scalarMapping",
147 attribute("INT_FROM_DEVICE_SCALAR", "INT/FROM_DEVICE_SCALAR",

155 BOOST_TEST(property.unit == "mm");
156 BOOST_TEST(property.attrDataFormat == TangoAdapter::AttrDataFormat::SCALAR);
157 BOOST_TEST(property.dataType == Tango::DEV_LONG);
158 BOOST_TEST(property.writeType == Tango::AttrWriteType::READ);
