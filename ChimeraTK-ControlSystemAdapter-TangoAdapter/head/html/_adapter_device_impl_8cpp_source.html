17 template<typename TangoType, typename AdapterType>
18 void writeInitialSpectrumValue(Tango::DeviceImpl* device, const std::string& memoriedValue,
19 Tango::WAttribute& writeAttribute, Tango::Attr& attr) {
20 // StringToArray will create a std::vector<bool> which is an optimised version of std::vector
21 // But Tango needs a plain array of bool, so we have to convert it here.
22 if constexpr(std::is_same_v<Tango::DevBoolean, TangoType>) {
23 auto values = stringToArray<TangoType>(memoriedValue);
24 // No lint, we cannot use std::array here because we do not know the size at compile time
25 // NOLINTNEXTLINE(modernize-avoid-c-arrays)
26 auto data = std::make_unique<Tango::DevBoolean[]>(values.size());
27 std::copy(values.begin(), values.end(), data.get());
28
29 // We can pass in the unique pointer here, set_write_value will copy the data
30 writeAttribute.set_write_value(data.get(), values.size());
31 }
32 else if constexpr(std::is_same_v<Tango::DevString, TangoType>) {
33 auto values = stringToArray<std::string>(memoriedValue);
34 // No lint, we cannot use std::array here because we do not know the size at compile time
35 // NOLINTNEXTLINE(modernize-avoid-c-arrays)
36 auto data = std::make_unique<Tango::DevString[]>(values.size());
37 std::transform(values.begin(), values.end(), data.get(), [&](auto& v) {
38 // NOLINTNEXTLINE
39 return const_cast<char*>(v.c_str());
40 });
41
42 // We can pass in the unique pointer and c_str() here, set_write_value will copy the data
43 // Lint: long is required by the Tango API
44 // NOLINTNEXTLINE(google-runtime-int)
45 writeAttribute.set_write_value(data.get(), static_cast<long>(values.size()));
46 }
47 else {
48 auto values = stringToArray<TangoType>(memoriedValue);
49 writeAttribute.set_write_value(values.data(), values.size());
50 }
51
52 attr.write(device, writeAttribute);
53 }
54} // namespace detail

127 void AdapterDeviceImpl::restoreMemoriedSpectra(const std::list<AttributeProperty>& writeableSpectrums) {
128 DEBUG_STREAM << AdapterDeviceClass::getClassName() << ": TangoAdapter::restoreMemoriedSpectra "
129 << writeableSpectrums.size() << std::endl;
130
131 // read spectrum values from memoried properties then write as initialised values
132 for(const auto& attProp : writeableSpectrums) {
133 DEBUG_STREAM << AdapterDeviceClass::getClassName() << ":name: " << attProp.name << " type:" << attProp.dataType
134 << std::endl;
135
136 // get write attribute name, in case it is different (READ_WITH_WRITE)
137 auto& write_attribute = get_device_attr()->get_w_attr_by_name(attProp.name.c_str());
138 std::string attName = attProp.name;
139 auto& baseAttribute = get_device_class()->get_class_attr()->get_attr(attName);
140
141 // get value of memoried property (__Memorized_<attributename>)
142 auto mem_value = getProperty<std::string>(this, "__Memoried_" + attProp.name);
143
144 DEBUG_STREAM << AdapterDeviceClass::getClassName() << ":__Memoried_" << attProp.name
145 << " mem_value: " << mem_value << std::endl;
146
147 if(mem_value.empty()) {
148 continue;
149 }
150
151 switch(attProp.dataType) {
152 case Tango::DEV_UCHAR: {
153 ::detail::writeInitialSpectrumValue<Tango::DevUChar, uint8_t>(
154 this, mem_value, write_attribute, baseAttribute);
155 break;
156 }
157
158 case Tango::DEV_ENUM: {
159 ::detail::writeInitialSpectrumValue<Tango::DevShort, int8_t>(this, mem_value, write_attribute, baseAttribute);
160 break;
161 }
162
163 case Tango::DEV_USHORT: {
164 ::detail::writeInitialSpectrumValue<Tango::DevUShort, uint16_t>(
165 this, mem_value, write_attribute, baseAttribute);
166 break;
167 }
168 case Tango::DEV_ULONG: {
169 ::detail::writeInitialSpectrumValue<Tango::DevULong, uint32_t>(
170 this, mem_value, write_attribute, baseAttribute);
171 break;
172 }
173 case Tango::DEV_ULONG64: {
174 ::detail::writeInitialSpectrumValue<Tango::DevULong64, uint64_t>(
175 this, mem_value, write_attribute, baseAttribute);
176 break;
177 }
178
179 case Tango::DEV_SHORT: {
180 ::detail::writeInitialSpectrumValue<Tango::DevShort, int16_t>(
181 this, mem_value, write_attribute, baseAttribute);
182 break;
183 }
184 case Tango::DEV_LONG: {
185 ::detail::writeInitialSpectrumValue<Tango::DevLong, int32_t>(this, mem_value, write_attribute, baseAttribute);
186 break;
187 }
188 case Tango::DEV_LONG64: {
189 ::detail::writeInitialSpectrumValue<Tango::DevLong64, int64_t>(
190 this, mem_value, write_attribute, baseAttribute);
191 break;
192 }
193 case Tango::DEV_FLOAT: {
194 ::detail::writeInitialSpectrumValue<Tango::DevFloat, float>(this, mem_value, write_attribute, baseAttribute);
195 break;
196 }
197
198 case Tango::DEV_DOUBLE: {
199 ::detail::writeInitialSpectrumValue<Tango::DevDouble, double>(
200 this, mem_value, write_attribute, baseAttribute);
201 break;
202 }
203 case Tango::DEV_BOOLEAN: {
204 ::detail::writeInitialSpectrumValue<Tango::DevBoolean, ChimeraTK::Boolean>(
205 this, mem_value, write_attribute, baseAttribute);
206 break;
207 }
208 case Tango::DEV_STRING: {
209 ::detail::writeInitialSpectrumValue<Tango::DevString, std::string>(
210 this, mem_value, write_attribute, baseAttribute);
211 break;
212 }
213 default:
214 ERROR_STREAM << AdapterDeviceClass::getClassName()
215 << ":TangoAdapter::restoreMemoriedSpectra - unknown datatype: " << attProp.dataType << std::endl;
216 }
217 }

222 void AdapterDeviceImpl::attachToClassAttributes(const std::shared_ptr<AttributeMapper::DeviceClass>& deviceClass) {
223 auto device = deviceClass->getDevice(get_name());
224 auto& adapter = TangoAdapter::getInstance();
225 auto csPvManager = adapter.getCsPvManager();
226 std::list<AttributeProperty> writeableSpectrums;
227
228 auto& updater = adapter.getUpdater();
229 DEBUG_STREAM << "Attaching Variables to device instance " << device->name << std::endl;
230 for(auto& attr : deviceClass->attributes) {
231 auto& source = device->attributeToSource[attr.name];
232 DEBUG_STREAM << " " << source << std::endl;
233
234 auto processVariable = csPvManager->getProcessVariable(source);
235 auto pv = ChimeraTK::TransferElementAbstractor(processVariable);
236 _attributeToPvMap[attr.name] = pv;
237
238 // Properly namespace the pv in the updater so we can distinguish per device
239 updater.addVariable(pv, get_name() + "/" + attr.name);
240
241 if(attr.attrDataFormat == AttrDataFormat::SPECTRUM &&
242 (attr.writeType == Tango::WRITE || attr.writeType == Tango::READ_WRITE)) {
243 writeableSpectrums.push_back(attr);
244 }
245 }
246
247 // only need to write spectrum attributes (manual writing bug in Tango)
248 // scalar attributes are memoried and initialized by Tango
249 // But only if we are running with a database. If not, there is nothing to
250 // restore anyway.
251 if(Tango::Util::_UseDb) {
252 restoreMemoriedSpectra(writeableSpectrums);
253 }
254
255 addMonitoredDeviceProperty(
256 device->stateSource, "state", "__ChimeraTK_toTangoState",
257 [](const std::type_info& type) {
258 bool integral = false;
259 ChimeraTK::callForTypeNoVoid(type, [&integral](auto t) {
260 if constexpr(std::is_integral_v<decltype(t)>) {
261 integral = true;
262 }
263 });
264 return integral;
265 },
266 [this](auto& pv) {
267 ChimeraTK::callForTypeNoVoid(pv.getValueType(), [this, &pv](auto t) {
268 // The value type was checked when the source was registered, so this branch can always be taken
269 if constexpr(std::is_integral_v<decltype(t)>) {
270 auto typedPv = boost::reinterpret_pointer_cast<ChimeraTK::NDRegisterAccessor<decltype(t)>>(
271 pv.getHighLevelImplElement());
272 // As per contract, this should be a StatusAccessor, so only 0, 1, 2 and 3 are supported and mapped
273 // properly. Everything else will be "UNKNOWN" and a message will be logged
274 switch(typedPv->accessData(0)) {
275 case 0:
276 set_state(Tango::DevState::ON);
277 break;
278 case 1:
279 set_state(Tango::DevState::FAULT);
280 break;
281 case 2:
282 set_state(Tango::DevState::OFF);
283 break;
284 case 3:
285 set_state(Tango::DevState::ALARM);
286 break;
287 default:
288 set_state(Tango::DevState::UNKNOWN);
289 TANGO_LOG_WARN << std::format(
290 "Trying to map {} to a TANGO state is not supported", typedPv->accessData(0));
291 break;
292 }
293 }
294 });
295 });
296
297 addMonitoredDeviceProperty(
298 device->statusSource, "status", "__ChimeraTK_toTangoStatus",
299 [](const std::type_info& type) { return type == typeid(std::string); },
300 [this](auto& pv) {
301 ChimeraTK::callForTypeNoVoid(pv.getValueType(), [this, &pv](auto t) {
302 // The value type was checked when the source was registered, so this branch can always be taken
303 if constexpr(std::is_same_v<decltype(t), std::string>) {
304 auto typedPv = boost::reinterpret_pointer_cast<ChimeraTK::NDRegisterAccessor<decltype(t)>>(
305 pv.getHighLevelImplElement());
306 set_status(typedPv->accessData(0));
307 }
308 });
309 });
310 }

314 void AdapterDeviceImpl::addMonitoredDeviceProperty(const std::optional<std::string>& source,
315 const std::string& purpose, const std::string& updaterName,
316 const std::function<bool(const std::type_info&)>& isSupportedType,
317 std::function<void(ChimeraTK::TransferElementAbstractor&)> map) {
318 if(!source) {
319 return;
320 }
321
322 auto& adapter = TangoAdapter::getInstance();
323
324 // The lookup throws a logic_error naming only the process variable when the source is not in the variable
325 // household of the application, so it is caught here and rethrown with the purpose it was configured for.
326 ChimeraTK::ProcessVariable::SharedPtr processVariable;
327 try {
328 processVariable = adapter.getCsPvManager()->getProcessVariable(source.value());
329 }
330 catch(ChimeraTK::logic_error& e) {
331 throw ChimeraTK::logic_error(std::format("{} to be used as device {}: {}", source.value(), purpose, e.what()));
332 }
333 if(!processVariable->isReadable()) {
334 throw ChimeraTK::logic_error(
335 std::format("{} to be used as device {}, but is not readable", source.value(), purpose));
336 }
337 if(!isSupportedType(processVariable->getValueType())) {
338 throw ChimeraTK::logic_error(
339 std::format("{} to be used as device {} has an unsupported value type", source.value(), purpose));
340 }
341
342 auto pv = ChimeraTK::TransferElementAbstractor(processVariable);
343
344 // The value which the application has already published and marked as valid when the device is created is
345 // applied as well, so the device does not keep the default state and status.
346 adapter.getUpdater().addVariable(pv, updaterName, [this, pv, map = std::move(map)]() mutable {
347 // A source which has never carried a value only holds an initialisation value which the application has not
348 // published, and a value which the application has marked as invalid must not be shown either.
349 if(pv.getVersionNumber() == ChimeraTK::VersionNumber(nullptr) ||
350 pv.dataValidity() != ChimeraTK::DataValidity::ok) {
351 return;
352 }
353
354 // This is called from the updater thread, so the device lock has to be taken before setting the state
355 // or the status
356 Tango::AutoTangoMonitor sync(this);
357 map(pv);
358 });
359 }
