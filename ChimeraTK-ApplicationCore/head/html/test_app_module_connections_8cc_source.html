117 BOOST_AUTO_TEST_CASE_TEMPLATE(testTwoScalarPushAccessors, T, test_types) {
118 // FIXME: With the new scheme, there cannot be a 1:1 module connection any more, it will always be a network
119 // involving the ControlSystem
120 std::cout << "*** testTwoScalarPushAccessors<" << typeid(T).name() << ">" << std::endl;
121
122 TestApplication<T> app;
123 app.testModuleFeed.feedingPush = {&app.testModuleFeed, "/testTwoScalarPushAccessors", "", ""};
124 app.testModuleConsume.consumingPush = {&app.testModuleConsume, "/testTwoScalarPushAccessors", "", ""};
125
126 ctk::TestFacility tf{app, false};
127 tf.runApplication();
128 app.testModuleFeed.mainLoopStarted.wait(); // make sure the module's mainLoop() is entered
129 app.testModuleConsume.mainLoopStarted.wait(); // make sure the module's mainLoop() is entered
130
131 // single threaded test
132 app.testModuleConsume.consumingPush = 0;
133 app.testModuleFeed.feedingPush = 42;
134 BOOST_CHECK(app.testModuleConsume.consumingPush == 0);
135 app.testModuleFeed.feedingPush.write();
136 BOOST_CHECK(app.testModuleConsume.consumingPush == 0);
137 app.testModuleConsume.consumingPush.read();
138 BOOST_CHECK(app.testModuleConsume.consumingPush == 42);
139
140 // launch read() on the consumer asynchronously and make sure it does not yet
141 // receive anything
142 auto futRead = std::async(std::launch::async, [&app] { app.testModuleConsume.consumingPush.read(); });
143 BOOST_CHECK(futRead.wait_for(std::chrono::milliseconds(200)) == std::future_status::timeout);
144
145 BOOST_CHECK(app.testModuleConsume.consumingPush == 42);
146
147 // write to the feeder
148 app.testModuleFeed.feedingPush = 120;
149 app.testModuleFeed.feedingPush.write();
150
151 // check that the consumer now receives the just written value
152 BOOST_CHECK(futRead.wait_for(std::chrono::milliseconds(2000)) == std::future_status::ready);
153 BOOST_CHECK(app.testModuleConsume.consumingPush == 120);
154 }

526 // Helper: return the <peer> elements of the given variable (found anywhere in
527 // the document), with the processed per-peer <unit>/<description> and the
528 // description priority attribute.
529 struct PeerInfo {
530 std::string name;
531 std::string direction;
532 std::string unit; // empty if no <unit> peer element
533 bool hasDescription{false};
534 std::string description;
535 int priority{0};
536 };
537
538 static std::vector<PeerInfo> getPeersForVariable(const std::string& xmlFile, const std::string& var) {
539 std::vector<PeerInfo> result;
540 xmlpp::DomParser parser;
541 parser.parse_file(xmlFile);
542 const auto* root = parser.get_document()->get_root_node();
543
544 // variables are nested arbitrarily deep in <module> elements, so a recursive traversal is used to find the
545 // one by name.
546 std::function<void(const xmlpp::Node*)> walk = [&](const xmlpp::Node* n) {
547 const auto* el = dynamic_cast<const xmlpp::Element*>(n);
548 if(el == nullptr) {
549 for(const auto* c = n->get_first_child(); c != nullptr; c = c->get_next_sibling()) {
550 walk(c);
551 }
552 return;
553 }
554 if(el->get_name() == "variable") {
555 const auto* an = el->get_attribute("name");
556 if(an != nullptr && an->get_value() == var) {
557 for(const auto* c = el->get_first_child(); c != nullptr; c = c->get_next_sibling()) {
558 const auto* cc = dynamic_cast<const xmlpp::Element*>(c);
559 if(cc == nullptr || cc->get_name() != "connections") {
560 continue;
561 }
562 for(const auto* p = cc->get_first_child(); p != nullptr; p = p->get_next_sibling()) {
563 const auto* pc = dynamic_cast<const xmlpp::Element*>(p);
564 if(pc == nullptr || pc->get_name() != "peer") {
565 continue;
566 }
567 PeerInfo pi;
568 if(auto* a = pc->get_attribute("name")) {
569 pi.name = a->get_value();
570 }
571 if(auto* a = pc->get_attribute("direction")) {
572 pi.direction = a->get_value();
573 }
574 for(const auto* ch = pc->get_first_child(); ch != nullptr; ch = ch->get_next_sibling()) {
575 const auto* chc = dynamic_cast<const xmlpp::Element*>(ch);
576 if(chc == nullptr) {
577 continue;
578 }
579 if(chc->get_name() == "unit") {
580 if(const auto* t = dynamic_cast<const xmlpp::TextNode*>(chc->get_first_child())) {
581 pi.unit = t->get_content();
582 }
583 }
584 else if(chc->get_name() == "description") {
585 pi.hasDescription = true;
586 if(auto* a = chc->get_attribute("priority")) {
587 pi.priority = std::stoi(a->get_value());
588 }
589 if(const auto* t = dynamic_cast<const xmlpp::TextNode*>(chc->get_first_child())) {
590 pi.description = t->get_content();
591 }
592 }
593 }
594 result.push_back(pi);
595 }
596 }
597 }
598 return; // a variable has no variable children
599 }
600 for(const auto* c = el->get_first_child(); c != nullptr; c = c->get_next_sibling()) {
601 walk(c);
602 }
603 };
604 walk(root);
605 return result;
606 }
