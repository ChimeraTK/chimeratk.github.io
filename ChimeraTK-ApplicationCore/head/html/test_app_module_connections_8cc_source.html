116 BOOST_AUTO_TEST_CASE_TEMPLATE(testTwoScalarPushAccessors, T, test_types) {
117 // FIXME: With the new scheme, there cannot be a 1:1 module connection any more, it will always be a network
118 // involving the ControlSystem
119 std::cout << "*** testTwoScalarPushAccessors<" << typeid(T).name() << ">" << std::endl;
120
121 TestApplication<T> app;
122 app.testModuleFeed.feedingPush = {&app.testModuleFeed, "/testTwoScalarPushAccessors", "", ""};
123 app.testModuleConsume.consumingPush = {&app.testModuleConsume, "/testTwoScalarPushAccessors", "", ""};
124
125 ctk::TestFacility tf{app, false};
126 tf.runApplication();
127 app.testModuleFeed.mainLoopStarted.wait(); // make sure the module's mainLoop() is entered
128 app.testModuleConsume.mainLoopStarted.wait(); // make sure the module's mainLoop() is entered
129
130 // single threaded test
131 app.testModuleConsume.consumingPush = 0;
132 app.testModuleFeed.feedingPush = 42;
133 BOOST_CHECK(app.testModuleConsume.consumingPush == 0);
134 app.testModuleFeed.feedingPush.write();
135 BOOST_CHECK(app.testModuleConsume.consumingPush == 0);
136 app.testModuleConsume.consumingPush.read();
137 BOOST_CHECK(app.testModuleConsume.consumingPush == 42);
138
139 // launch read() on the consumer asynchronously and make sure it does not yet
140 // receive anything
141 auto futRead = std::async(std::launch::async, [&app] { app.testModuleConsume.consumingPush.read(); });
142 BOOST_CHECK(futRead.wait_for(std::chrono::milliseconds(200)) == std::future_status::timeout);
143
144 BOOST_CHECK(app.testModuleConsume.consumingPush == 42);
145
146 // write to the feeder
147 app.testModuleFeed.feedingPush = 120;
148 app.testModuleFeed.feedingPush.write();
149
150 // check that the consumer now receives the just written value
151 BOOST_CHECK(futRead.wait_for(std::chrono::milliseconds(2000)) == std::future_status::ready);
152 BOOST_CHECK(app.testModuleConsume.consumingPush == 120);
153 }

525 // Helper: return the <peer> elements of the given variable (found anywhere in
526 // the document), with the processed per-peer <unit>/<description> and the
527 // description priority attribute.
528 struct PeerInfo {
529 std::string name;
530 std::string direction;
531 std::string unit; // empty if no <unit> peer element
532 bool hasDescription{false};
533 std::string description;
534 int priority{0};
535 };
536
537 static std::vector<PeerInfo> getPeersForVariable(const std::string& xmlFile, const std::string& var) {
538 std::vector<PeerInfo> result;
539 xmlpp::DomParser parser;
540 parser.parse_file(xmlFile);
541 const auto* root = parser.get_document()->get_root_node();
542
543 // variables are nested arbitrarily deep in <module> elements, so a recursive traversal is used to find the
544 // one by name.
545 std::function<void(const xmlpp::Node*)> walk = [&](const xmlpp::Node* n) {
546 const auto* el = dynamic_cast<const xmlpp::Element*>(n);
547 if(el == nullptr) {
548 for(const auto* c = n->get_first_child(); c != nullptr; c = c->get_next_sibling()) {
549 walk(c);
550 }
551 return;
552 }
553 if(el->get_name() == "variable") {
554 const auto* an = el->get_attribute("name");
555 if(an != nullptr && an->get_value() == var) {
556 for(const auto* c = el->get_first_child(); c != nullptr; c = c->get_next_sibling()) {
557 const auto* cc = dynamic_cast<const xmlpp::Element*>(c);
558 if(cc == nullptr || cc->get_name() != "connections") {
559 continue;
560 }
561 for(const auto* p = cc->get_first_child(); p != nullptr; p = p->get_next_sibling()) {
562 const auto* pc = dynamic_cast<const xmlpp::Element*>(p);
563 if(pc == nullptr || pc->get_name() != "peer") {
564 continue;
565 }
566 PeerInfo pi;
567 if(auto* a = pc->get_attribute("name")) {
568 pi.name = a->get_value();
569 }
570 if(auto* a = pc->get_attribute("direction")) {
571 pi.direction = a->get_value();
572 }
573 for(const auto* ch = pc->get_first_child(); ch != nullptr; ch = ch->get_next_sibling()) {
574 const auto* chc = dynamic_cast<const xmlpp::Element*>(ch);
575 if(chc == nullptr) {
576 continue;
577 }
578 if(chc->get_name() == "unit") {
579 if(const auto* t = dynamic_cast<const xmlpp::TextNode*>(chc->get_first_child())) {
580 pi.unit = t->get_content();
581 }
582 }
583 else if(chc->get_name() == "description") {
584 pi.hasDescription = true;
585 if(auto* a = chc->get_attribute("priority")) {
586 pi.priority = std::stoi(a->get_value());
587 }
588 if(const auto* t = dynamic_cast<const xmlpp::TextNode*>(chc->get_first_child())) {
589 pi.description = t->get_content();
590 }
591 }
592 }
593 result.push_back(pi);
594 }
595 }
596 }
597 return; // a variable has no variable children
598 }
599 for(const auto* c = el->get_first_child(); c != nullptr; c = c->get_next_sibling()) {
600 walk(c);
601 }
602 };
603 walk(root);
604 return result;
605 }
