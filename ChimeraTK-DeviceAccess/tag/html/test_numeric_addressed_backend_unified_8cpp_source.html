193template<typename Derived, typename rawUserType>
194struct ShortRaw_base {
195 Derived* derived{static_cast<Derived*>(this)};
196
197 bool isWriteable() { return true; }
198 bool isReadable() { return true; }
199 ChimeraTK::AccessModeFlags supportedFlags() { return {ChimeraTK::AccessMode::raw}; }
200 size_t nChannels() { return 1; }
201 size_t nElementsPerChannel() { return 1; }
202 size_t writeQueueLength() { return std::numeric_limits<size_t>::max(); }
203 size_t nRuntimeErrorCases() { return 1; }
204
205 // The padding (i.e. the bits of the 32 bit word which are not part of the register under test but of the neighbouring
206 // registers) needs to be checked to be unchanged in write operations. This test assumes that in between the calls to
207 // generateValue() and getRemoteValue() the padding bits are not *intentionally* changed (e.g. by testing another
208 // register in between). This is true for the UnifiedBackendTest right now.
209 // TODO: In future it may be better to foresee a "no-side-effect" test in the UnifiedBackendTest, which tests that
210 // other registers stay unchanged (caveat: need to specify which registers to look at! We have intentionally
211 // overlapping registers in this test which of course will change!)
212 int32_t lastPadding{0};
213 bool printPaddingChangedMessage{true};
214
215 static constexpr auto capabilities = TestCapabilities<>()
216 .disableForceDataLossWrite()
217 .disableAsyncReadInconsistency()
218 .disableSwitchReadOnly()
219 .disableSwitchWriteOnly()
220 .disableTestWriteNeverLosesData()
221 .enableTestRawTransfer();
222
223 // This register shares the address space with all our test registers. It gives us direct access to the 4 byte
224 // address range, so we can test the correct placement of the unaligned values.
225 DummyRegisterRawAccessor acc{exceptionDummy, "", "/Integers/unsigned32"};
226
227 rawUserType get() { return (acc & derived->bitmask) >> derived->bitshift; }
228
229 void set(rawUserType val) {
230 acc &= ~derived->bitmask;
231 lastPadding = acc;
232 printPaddingChangedMessage = true;
233 acc |= (val << derived->bitshift) & derived->bitmask;
234 }
235
236 // Type can be user type or raw type, depeding of the raw flag
237 template<typename Type>
238 std::vector<std::vector<Type>> generateValue(bool raw = false) {
239 rawUserType newRawValue = static_cast<rawUserType>(get() + derived->rawIncrement);
240 Type v = ChimeraTK::numeric::convert<Type>(raw ? newRawValue : (newRawValue * derived->rawToCooked));
241 /* std::cout << "generateValue " << derived->path() << " " << float(rawUserType(get() + derived->rawIncrement))
242 << " -> " << float(v) << std::endl; */
243 lastPadding = acc & ~derived->bitmask;
244 printPaddingChangedMessage = true;
245 return {{v}};
246 }
247
248 // We use the same function to implement raw and cooked reading.
249 // Do the calculation on the target type, then decide on the padding error.
250 // rawToCooked might undo the ++ effect in case
251 // of rounding. When getting cooked, the ++ must happen on the UserType.
252 // Type can be UserType or RawType.
253 template<typename Type>
254 std::vector<std::vector<Type>> getRemoteValue(bool raw = false) {
255 Type v = get() * (raw ? 1 : derived->rawToCooked);
256 if((acc & ~derived->bitmask) != lastPadding) {
257 if(printPaddingChangedMessage) {
258 std::cerr << "getRemoteValue(): Padding data has changed. Test will be failed by returing a false remote value "
259 << "(off by one)." << std::endl;
260 printPaddingChangedMessage = false;
261 }
262 v++;
263 return {{v}};
264 }
265 /* std::cout << "getRemoteValue " << derived->path() << " " << float(get()) << " -> " << float(v) << std::endl; */
266 return {{v}};
267 }
268
269 void setRemoteValue() {
270 set(get() + derived->rawIncrement);
271 /* std::cout << "setRemoteValue " << derived->path() << " " << float(get()) << " -> " <<
272 getRemoteValue<float>()[0][0]
273 << std::endl; */
274 }

276 void setForceRuntimeError(bool enable, size_t) {
277 exceptionDummy->throwExceptionRead = enable;
278 exceptionDummy->throwExceptionWrite = enable;
279 exceptionDummy->throwExceptionOpen = enable;
280 }
281};

887 template<typename Type>
888 std::vector<std::vector<Type>> generateValue(bool getRaw = false) {
889 std::vector<Type> v;
890 typedef typename Derived::rawUserType Traw;
891 typedef typename Derived::minimumUserType T;
892 auto cv = derived->template getRemoteValue<Traw>(true)[0];
893 for(size_t i = 0; i < derived->nElementsPerChannel(); ++i) {
894 Traw e = cv[i] + derived->increment * (static_cast<Traw>(i) + 1);
895 if(!getRaw) {
896 v.push_back(derived->template convertRawToCooked<T, Traw>(e));

908 typedef typename Derived::rawUserType Traw;
909
910 std::vector<UserType> cookedValues(derived->nElementsPerChannel());
911 std::vector<Traw> rawValues(derived->nElementsPerChannel());
912
913 // Keep the scope of the dummy buffer lock as limited as possible (see #12332).
914 // The rawToCooked conversion will acquire a lock via the math plugin decorator,
915 // which will cause lock order inversion if you hold the dummy buffer lock at that point.
916 {
917 auto bufferLock = derived->acc.getBufferLock();
918
919 for(size_t i = 0; i < derived->nElementsPerChannel(); ++i) {
920 rawValues[i] = derived->acc[i + derived->myOffset()];
921 }
922 } // end of bufferLock scope
923
924 for(size_t i = 0; i < derived->nElementsPerChannel(); ++i) {
925 if(!getRaw) {
926 cookedValues[i] = derived->template convertRawToCooked<UserType, Traw>(rawValues[i]);
927 }
928 else {
929 cookedValues[i] = static_cast<UserType>(rawValues[i]);
930 }
931 }
932 return {cookedValues};
933 }

937 { // scope for the buffer lock
938 auto bufferLock = derived->acc.getBufferLock();
939 for(size_t i = 0; i < derived->nElementsPerChannel(); ++i) {
940 derived->acc[i + derived->myOffset()] = v[i];
941 }
942 } // release the buffer lock before triggering another thread
943 if(derived->isPush()) {
944 dynamic_cast<ExceptionDummy&>(derived->acc.getBackend()).triggerInterrupt(6);
945 }
946 }

1384// Bit ranges inside the named channels of a 2D register (bitRangeChannels.jmap). A channel carrying bit-field
1385// children (A) exposes a full-word parent slice plus one bit-range slice per child; a channel that is itself a
1386// single bit range (B) exposes a bit-range slice. All slices read the strided 2D sample data correctly. Writes to
1387// a bit-range slice are rejected and obtaining the full-2D accessor of a register with a direct bit-range channel
1388// raises a clear error.
1389BOOST_AUTO_TEST_CASE(TestBitRangeNamedChannelSlices) {
1390 std::cout << "*** TestBitRangeNamedChannelSlices *** " << std::endl;
1391

1394
1395 // Place known sample words into the shared DMA buffer through the plain read-write view RAW (BAR 13, address
1396 // 0x4000). Each 64-bit element word of TEST/BR spans two RAW words: the lower one is channel A's word at byte
1397 // offset 0x4000 + 8*e, the upper one is channel B's word at byte offset 0x4004 + 8*e.
1398 DummyRegisterAccessor<uint32_t> raw{dummyBitRangeChannels.get(), "RAW", ""};

1403 wordA[e] = 0x12345678u + e * 0x01010101u;
1404 wordB[e] = 0xABCDEF01u + e * 0x00001111u;
1405 raw[2 * e] = wordA[e];
