54static bool runTool(
55 const std::string& extraArgs, const std::string& stdoutFile, const std::string& stderrFile, int& exitCode) {
56 std::string command =
57 std::format(R"("{}" {} >"{}" 2>"{}")", CHIMERATK_DEVICE_TO_JMAP, extraArgs, stdoutFile, stderrFile);
58 int status = std::system(command.c_str());
59 if(!WIFEXITED(status)) {
60 return false;
61 }
62 exitCode = WEXITSTATUS(status);
63 return true;
64}

69static std::string readFile(const std::string& path) {

71 std::ostringstream buffer;
72 buffer << stream.rdbuf();
73 return buffer.str();

85static bool compareRegisters(const RegisterInfo& source, const RegisterInfo& destination) {
86 if(source.getNumberOfElements() != destination.getNumberOfElements()) {
87 return false;
88 }
89 const DataDescriptor& sourceDescriptor = source.getDataDescriptor();
90 const DataDescriptor& destinationDescriptor = destination.getDataDescriptor();
91 if(sourceDescriptor.fundamentalType() != destinationDescriptor.fundamentalType()) {
92 return false;
93 }
94 // The read capability must match.
95 if(source.isReadable() != destination.isReadable()) {
96 return false;
97 }
98 // The only accepted difference in write capability is a writable source register becoming read-only (the documented
99 // lossy transformation of writable wait_for_new_data registers into read-only interrupts).
100 if(source.isWriteable() != destination.isWriteable() && (!source.isWriteable() || destination.isWriteable())) {
101 return false;
102 }
103 // Signedness, integrality and element size come from DataDescriptor::minimumDataType() and are only defined for
104 // numeric registers.
105 if(sourceDescriptor.fundamentalType() == DataDescriptor::FundamentalType::numeric) {
106 return sourceDescriptor.isSigned() == destinationDescriptor.isSigned() &&
107 sourceDescriptor.isIntegral() == destinationDescriptor.isIntegral() &&
108 sourceDescriptor.minimumDataType().getNumberOfBytes() ==
109 destinationDescriptor.minimumDataType().getNumberOfBytes();
110 }
111 return true;
112}

122static size_t compareCatalogues(const RegisterCatalogue& source, const RegisterCatalogue& destination) {
123 size_t mismatches = 0;
124 for(const auto& registerInfo : source) {
125 if(registerInfo.getNumberOfDimensions() > 1) {
126 continue;
127 }
128 const std::string path = registerInfo.getRegisterName();
129 if(!destination.hasRegister(path)) {
130 ++mismatches;
131 continue;
132 }
133 if(!compareRegisters(source.getRegister(path), destination.getRegister(path))) {
134 ++mismatches;
135 }
136 }
137 // Every one-dimensional destination register must also be present in the source.
138 for(const auto& registerInfo : destination) {
139 if(registerInfo.getNumberOfDimensions() > 1) {
140 continue;
141 }
142 const std::string path = registerInfo.getRegisterName();
143 if(!source.hasRegister(path)) {
144 ++mismatches;
145 }
146 }
147 return mismatches;
148}

153static size_t roundTrip(const std::string& sourceCdd, const std::string& toolArguments, const std::string& outputFile,
154 const std::string& stdoutFile, const std::string& stderrFile) {
155 Device source;
156 source.open(sourceCdd);
157 auto sourceCatalogue = source.getRegisterCatalogue();
158
159 int exitCode = -1;
160 BOOST_REQUIRE(runTool(toolArguments, stdoutFile, stderrFile, exitCode));
161 BOOST_REQUIRE(exitCode == 0);
162
163 Device destination;
164 destination.open("(dummy?map=" + outputFile + ")");
165 auto destinationCatalogue = destination.getRegisterCatalogue();
166
167 return compareCatalogues(sourceCatalogue, destinationCatalogue);
168}

214static std::set<std::string> collectWritableInterrupts(const RegisterCatalogue& catalogue) {
215 std::set<std::string> result;
216 for(const auto& registerInfo : catalogue) {
217 if(registerInfo.isWriteable() &&
218 registerInfo.getSupportedAccessModes().has(ChimeraTK::AccessMode::wait_for_new_data) &&
219 registerInfo.getDataDescriptor().fundamentalType() != DataDescriptor::FundamentalType::nodata) {
220 result.insert(registerInfo.getRegisterName());
221 }
222 }
223 return result;
224}
225
226/**********************************************************************************************************************/
227
229static std::set<std::string> collectTwoDimensional(const RegisterCatalogue& catalogue) {
230 std::set<std::string> result;
231 for(const auto& registerInfo : catalogue) {
232 if(registerInfo.getNumberOfDimensions() > 1) {
233 result.insert(registerInfo.getRegisterName());
234 }
235 }
236 return result;
237}
