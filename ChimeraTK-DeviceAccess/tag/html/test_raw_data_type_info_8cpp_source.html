15// DataType::getNumberOfBytes() returns the element data width in bytes for every DataType value: 1/2/4/8 for the
16// integer types by their width, 4 for float32 and 8 for float64, sizeof(ChimeraTK::Boolean) for Boolean, 0 for
17// none and Void (no data). We cannot test string, as it will trigger an assertion.
18BOOST_AUTO_TEST_CASE(TestGetNumberOfBytes) {
19 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::int8).getNumberOfBytes()) == 1);
20 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::uint8).getNumberOfBytes()) == 1);
21 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::int16).getNumberOfBytes()) == 2);
22 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::uint16).getNumberOfBytes()) == 2);
23 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::int32).getNumberOfBytes()) == 4);
24 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::uint32).getNumberOfBytes()) == 4);
25 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::int64).getNumberOfBytes()) == 8);
26 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::uint64).getNumberOfBytes()) == 8);
27 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::float32).getNumberOfBytes()) == 4);
28 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::float64).getNumberOfBytes()) == 8);
29 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::Boolean).getNumberOfBytes()) == sizeof(ChimeraTK::Boolean));
30 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::none).getNumberOfBytes()) == 0);
31 BOOST_TEST((ChimeraTK::DataType(ChimeraTK::DataType::Void).getNumberOfBytes()) == 0);
32}
