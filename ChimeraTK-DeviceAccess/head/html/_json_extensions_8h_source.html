75 template<typename... Ts>
76 // NOLINTNEXTLINE(readability-identifier-naming) - the name is mandated by nlohmann
77 void adl_serializer<std::variant<Ts...>>::from_json(const json& j, std::variant<Ts...>& var) {
78 bool parsed = false;
79 // Try the alternatives in declaration order and keep the first one that parses, so the mapping is independent of
80 // which alternatives happen to be ambiguous.
81 auto tryParse = [&](const auto& value) {
82 using T = std::decay_t<decltype(value)>;
83 if(parsed) {
84 return;
85 }
86 try {
87 // Skip a null explicitly: nlohmann's from_json for POD types accepts null and leaves the value default, so it
88 // would silently select the first alternative instead of reporting the malformed input. A variant always holds
89 // a value, so no alternative reads a null.
90 if(j.is_null()) {
91 return;
92 }
93 var = j.get<T>();
94 parsed = true;
95 }
96 catch(const json::exception&) {
97 }
98 };
99 (tryParse(Ts{}), ...);
100 if(!parsed) {
101 throw json::type_error::create(302, "No alternative of the variant matches the JSON value.", &j);
102 }
103 }
