4"""jmap formatter.
5
6Re-formats ChimeraTK .jmap files. The syntax is JSON, so base formatting follows standard
7JSON rules (2-space indent, ': ' and ', ' separators). On top of that, well-known *leaf*
8objects are compacted onto the fewest lines that fit within MAX_LINE_LENGTH: a single
9line when the whole object fits, otherwise its scalar members are packed onto as few
10lines as possible (never exceeding 120 chars). Empty objects/arrays render as '{}'/'[]'
11and non-ASCII characters are preserved verbatim.
12
13Compacted object keys (cannot have children in the schema):
14 address, representation, selectedBy, secondaryBufferAddress, INTC
15Compacted array keys (flat lists of scalars, compacted when they fit):
16 options, triggeredByInterrupt
17
18MAX_LINE_LENGTH (120 chars) is the overriding rule: nothing, including the compacted
19object/array keys above, is allowed to exceed it.
20
21Usage:
22 python3 jmapfmt.py fix <files...> # rewrite files in place
23 python3 jmapfmt.py check <files...> # exit 1 if any file is not formatted (no writes)
24 python3 jmapfmt.py check < list_of_files # read the file list from stdin
25 python3 jmapfmt.py stdin # read from stdin, write formatted text to stdout
26"""

34# Object keys whose value is a schema leaf (no children): render on one line if it fits.
35COMPACT_OBJECT_KEYS = {"address", "representation", "selectedBy", "secondaryBufferAddress", "INTC"}
36# Array keys whose value is a flat list of scalars: render on one line if it fits.
37COMPACT_ARRAY_KEYS = {"options", "triggeredByInterrupt"}

40def _scalar(value):
41 if isinstance(value, bool):
42 return "true" if value else "false"
43 if value is None:
44 return "null"
45 return json.dumps(value, ensure_ascii=False)

48def _inline(value):
49 """Render a value on a single (compact) line, e.g. {'a': 1, 'b': 2} or [1, 2, 3]."""
50 if isinstance(value, dict):
51 if not value:
52 return "{}"
53 return "{" + ", ".join(f"{json.dumps(k, ensure_ascii=False)}: {_inline(v)}" for k, v in value.items()) + "}"
54 if isinstance(value, list):
55 if not value:
56 return "[]"
57 return "[" + ", ".join(_inline(v) for v in value) + "]"
58 return _scalar(value)

61def _ind(n):

65def _obj_body(indent, members):
66 """Lines for the body of an object (no outer braces), with commas between members."""
67 groups = []
68 for k, v in members:
69 if isinstance(v, (dict, list)):
70 groups.append(_keyed_member_lines(indent, k, v))
71 else:
72 groups.append([_ind(indent) + f"{json.dumps(k, ensure_ascii=False)}: {_scalar(v)}"])
73 out = []
74 for i, g in enumerate(groups):
75 out.extend(g)
76 if i < len(groups) - 1:
77 out[-1] = out[-1] + ","
78 return out

81def _arr_body(indent, items):
82 """Lines for the body of an array (no outer brackets), with commas between items."""
83 groups = []
84 for item in items:
85 if isinstance(item, dict):
86 if not item:
87 g = [_ind(indent) + "{}"]
88 else:
89 g = [_ind(indent) + "{"] + _obj_body(indent + 1, list(item.items())) + [_ind(indent) + "}"]
90 elif isinstance(item, list):
91 if not item:
92 g = [_ind(indent) + "[]"]
93 else:
94 g = [_ind(indent) + "["] + _arr_body(indent + 1, item) + [_ind(indent) + "]"]
95 else:
96 g = [_ind(indent) + _scalar(item)]
97 groups.append(g)
98 out = []
99 for i, g in enumerate(groups):
100 out.extend(g)
101 if i < len(groups) - 1:
102 out[-1] = out[-1] + ","
103 return out

106def _obj_body_packed(indent, members):
107 """Lines for the body of a compact leaf object that does not fit on one line.
108
109 Members are packed onto as few lines as possible, with each line (including the
110 current indentation) kept within MAX_LINE_LENGTH. This avoids the all-or-nothing
111 behaviour of fully expanding every member once the single-line form overflows.
112 """
113 pad = _ind(indent)
114 pieces = []
115 for k, v in members:
116 kk = json.dumps(k, ensure_ascii=False)
117 if isinstance(v, (dict, list)):
118 pieces.append(f"{kk}: {_inline(v)}")
119 else:
120 pieces.append(f"{kk}: {_scalar(v)}")
121 lines = []
122 current = ""
123 for piece in pieces:
124 candidate = (current + ", " + piece) if current else piece
125 if len(pad) + len(candidate) <= MAX_LINE_LENGTH:
126 current = candidate
127 else:
128 lines.append(current)
129 current = piece
130 if current:
131 lines.append(current)
132 out = []
133 for i, line in enumerate(lines):
134 out.append(pad + line + ("," if i < len(lines) - 1 else ""))
135 return out

138def _keyed_member_lines(indent, key, value):
139 """Lines for `key: <dict|list>`, first line carries the key, no trailing comma.
140
141 Known leaf objects are compacted: if the whole object fits MAX_LINE_LENGTH it is
142 rendered on a single line, otherwise its scalar members are packed onto the fewest
143 lines that stay within MAX_LINE_LENGTH. Short scalar arrays are always compacted.
144 """
145 pad = _ind(indent)
146 if isinstance(value, dict):
147 if not value:
148 return [pad + f"{json.dumps(key, ensure_ascii=False)}: {{}}"]
149 if key in COMPACT_OBJECT_KEYS:
150 line = f"{json.dumps(key, ensure_ascii=False)}: {_inline(value)}"
151 if len(line) <= MAX_LINE_LENGTH - 2 * indent:
152 return [pad + line]
153 # Too long for a single line: pack the members greedily, each line <= MAX_LINE_LENGTH.
154 return [pad + f"{json.dumps(key, ensure_ascii=False)}: {{"] + _obj_body_packed(
155 indent + 1, list(value.items())) + [pad + "}"]
156 return [pad + f"{json.dumps(key, ensure_ascii=False)}: {{"] + _obj_body(indent + 1, list(value.items())) + [pad + "}"]
157 # list
158 if not value:
159 return [pad + f"{json.dumps(key, ensure_ascii=False)}: []"]
160 if key in COMPACT_ARRAY_KEYS:
161 line = f"{json.dumps(key, ensure_ascii=False)}: {_inline(value)}"
162 if len(line) <= MAX_LINE_LENGTH - 2 * indent:
163 return [pad + line]
164 # The 120-char rule takes precedence: fall back to expanded form.
165 return [pad + f"{json.dumps(key, ensure_ascii=False)}: ["] + _arr_body(indent + 1, value) + [pad + "]"]

168def format_text(text):
169 """Return the formatted representation of a jmap/JSON document, or None if unparseable."""
170 try:
171 data = json.loads(text)
172 except json.JSONDecodeError:
173 return None
174 out = ["{"]
175 out += _obj_body(1, list(data.items()))
176 out.append("}")
177 return "\n".join(out) + "\n"

190def main():
191 parser = argparse.ArgumentParser(
192 prog="jmapfmt",
193 description="Re-format ChimeraTK .jmap files.",
194 epilog="Examples:\n"
195 " jmapfmt fix a.jmap b.jmap # rewrite in place\n"
196 " jmapfmt check a.jmap # verify, exit 1 if not formatted\n"
197 " find . -name '*.jmap' | jmapfmt check\n"
198 " jmapfmt stdin < a.jmap # write formatted text to stdout",
199 formatter_class=argparse.RawDescriptionHelpFormatter)
200 sub = parser.add_subparsers(dest="command", required=True, metavar="{fix,check,stdin}")
201
202 p_fix = sub.add_parser("fix", help="rewrite files in place",
203 description="Rewrite the given files in place, adding or fixing formatting.")
204 p_fix.add_argument("files", nargs="+", metavar="FILE",
205 help="the .jmap files to rewrite")
206
207 p_check = sub.add_parser("check", help="verify formatting, no writes; exit 1 on mismatch",
208 description="Verify that the given files are formatted. "
209 "No files are modified; exits 1 if any file differs. "
210 "If no files are given, paths are read from stdin, one per line.")
211 p_check.add_argument("files", nargs="*", metavar="FILE",
212 help="the .jmap files to check; if omitted, read paths from stdin")
213
214 p_stdin = sub.add_parser("stdin", help="read jmap text from stdin, write formatted text to stdout",
215 description="Read a jmap document from stdin and write the formatted "
216 "result to stdout. Exits 1 on unparseable input.")
217
218 args = parser.parse_args()
219
220 if args.command == "stdin":
221 text = sys.stdin.read()
222 formatted = format_text(text)
223 if formatted is None:
224 print("jmapfmt: ERROR: stdin is not valid JSON", file=sys.stderr)
225 sys.exit(1)
226 sys.stdout.write(formatted)
227 sys.exit(0)
228
229 if args.command == "check" and not args.files:
230 args.files = [line.strip() for line in sys.stdin if line.strip()]
231
232 bad = 0
233 for path in args.files:
234 original = _read_file(path)
235 formatted = format_text(original)
236 if formatted is None:
237 print(f"jmapfmt: ERROR: could not parse {path} (invalid JSON)", file=sys.stderr)
238 bad = 1
239 continue
240 if args.command == "check":
241 if original != formatted:
242 print(f"jmapfmt: {path} is not formatted")
243 bad = 1
244 else: # fix
245 if original != formatted:
246 _write_file(path, formatted)
247 print(f"jmapfmt: formatted {path}")
248 sys.exit(1 if bad else 0)
