35static CatalogueSnapshot snapshot(const NumericAddressedRegisterCatalogue& catalogue) {
36 CatalogueSnapshot result;
37 for(const auto& info : catalogue) {
38 result.registerPaths.insert(info.getRegisterName());
39 }
40 result.interrupts = catalogue.getListOfInterrupts();
41 return result;
42}
43
44static void checkUnchanged(const CatalogueSnapshot& before, const CatalogueSnapshot& after) {
45 BOOST_TEST(after.registerPaths == before.registerPaths);
46 BOOST_TEST(after.interrupts == before.interrupts);
47}

53// The dispatcher is a no-op for a controller type it does not know and for the built-in "dummy" type, which has no
54// augmenter. Both must leave the catalogue untouched.
55BOOST_AUTO_TEST_CASE(testAugmentCatalogueNoOp) {

76// Creating a distributor for a controller type that is not registered throws, even though the dispatcher (which
77// shares the record lookup) skips it silently. The type lookup happens before any ISR is opened, so a backend whose
78// metadata declares an unknown type is enough.
79BOOST_AUTO_TEST_CASE(testCreateUnknownControllerTypeThrows) {
80 // A minimal map: metadata declares an unknown controller type for interrupt [0] and one register carries an
81 // interrupt of the sub-handler [0, 0]. Requesting that sub-interrupt forces the creation of a distributor for
82 // handler [0], which resolves the unregistered type.
83 const std::string mapFile = "factoryUnknownController_" + std::to_string(getpid()) + ".mapp";
84 {
85 std::ofstream out(mapFile);
86 out << R"(@![0] {"MyUnknownController":{"module":"/ctrl0"}})" << "\n"
87 << R"(/ctrl0/active 1 0 4 0 32 0 0 RW)" << "\n"
88 << R"(/myinterrupt 0 0 0 0 0 0 0 INTERRUPT0:0)" << "\n";
89 }
90
91 Device device{"(dummy?map=" + mapFile + ")"};
92 device.open();
93 device.activateAsyncRead();
94
95 BOOST_CHECK_THROW(
96 (void)device.getVoidRegisterAccessor("!0:0", {ChimeraTK::AccessMode::wait_for_new_data}), ChimeraTK::logic_error);
97
98 device.close();
99 std::remove(mapFile.c_str());
100}

104// The descriptor validation of getInterruptControllerNameAndDescription also guards the metadata catalogue path, which
105// a traditional map file uses: the metadata carries the controller descriptor verbatim and never passes through the
106// JSON entry reader. A descriptor with several controller keys is a malformed entry and must make the distributor
107// creation throw rather than silently picking the first key.
108//
109// The map is a complete INTC register set for handler [0] (its ISR, the interface-enable register and a 1-bit child of
110// ISR), so with the check dropped the entry would resolve the first controller key to a fully usable "INTC"
111// distributor and the interrupt accessor would be created successfully. The map deliberately does not lean on the
112// register lookup to fail for an unrelated reason, so the test fails exactly when the descriptor check is absent.
113BOOST_AUTO_TEST_CASE(testCreateSeveralControllerKeysThrows) {
114 const std::string mapFile = "factorySeveralControllerKeys_" + std::to_string(getpid()) + ".mapp";
115 {
116 std::ofstream out(mapFile);
117 out << R"(@![0] {"INTC":{"path":"TEST0","version":1},"OTHER":{"path":"TEST0"}})" << "\n"
118 << R"(TEST0.IER 1 0x8000C 4 0 7 0 0 RW)" << "\n"
119 << R"(TEST0.ISR 1 0x80008 4 0 7 0 0 RW)" << "\n"
120 << R"(TEST0.DAQ_READY 0 0 0 0 0 0 0 INTERRUPT0:4)" << "\n";
121 }
122
123 BOOST_CHECK_THROW(
124 [&] {
125 Device device{"(dummy?map=" + mapFile + ")"};
126 device.open();
127 device.activateAsyncRead();
128 (void)device.getVoidRegisterAccessor("!0:4", {ChimeraTK::AccessMode::wait_for_new_data});
129 }(),
130 ChimeraTK::logic_error);
131
132 std::remove(mapFile.c_str());
133}
