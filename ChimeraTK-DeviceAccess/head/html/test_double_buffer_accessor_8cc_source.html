15// Test that a full read cycle completes successfully.
16// Observable behaviour: after read(), the accessor holds data
17// and the enable register is left in the enabled state (value 1).
18BOOST_AUTO_TEST_CASE(test_full_read_cycle) {
19 Device device;
20 device.open("(dummy?map=simpleJsonFile.jmap)");
21
22 // Get double-buffer accessor through the normal public API.
23 // DAQ.MACRO_PULSE_NUMBER is a simple 1-element uint32 register
24 // which uses double-buffering with index 1.
25 auto accessor = device.getOneDRegisterAccessor<int>("/DAQ/MACRO_PULSE_NUMBER", 1, 0);
26
27 // Perform a full read cycle through the public API
28 accessor.read();
29
30 // After the read cycle, we have valid data
31 BOOST_CHECK(accessor.dataValidity() != ChimeraTK::DataValidity::faulty);
32}

34// ------------------------------------------------------------
35// Test that concurrent reads on the same double-buffered register
36// complete successfully. The transfer lock ensures sequential
37// access to the handshake registers, so no data corruption occurs.
38BOOST_AUTO_TEST_CASE(test_concurrent_reads_same_register) {
39 Device device;
40 device.open("(dummy?map=simpleJsonFile.jmap)");
41
42 // Two accessors on the same register
43 auto accessor1 = device.getOneDRegisterAccessor<int>("/DAQ/MACRO_PULSE_NUMBER", 1, 0);
44 auto accessor2 = device.getOneDRegisterAccessor<int>("/DAQ/MACRO_PULSE_NUMBER", 1, 0);
45
46 // Read concurrently from two threads. Both must complete without
47 // deadlock or data corruption.
48 auto future = std::async(std::launch::async, [&] { accessor1.read(); });
49 accessor2.read();
50
51 auto status = future.wait_for(std::chrono::milliseconds(200));
52 BOOST_CHECK(status == std::future_status::ready);
53}
