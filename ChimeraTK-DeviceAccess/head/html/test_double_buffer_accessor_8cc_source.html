16// Test that a full read cycle completes successfully.
17// Observable behaviour: after read(), the accessor holds data
18// and the enable register is left in the enabled state (value 1).
19BOOST_AUTO_TEST_CASE(test_full_read_cycle) {
20 Device device;
21 device.open("(dummy?map=simpleJsonFile.jmap)");
22
23 // Get double-buffer accessor through the normal public API.
24 // DAQ.MACRO_PULSE_NUMBER is a simple 1-element uint32 register
25 // which uses double-buffering with index 1.
26 auto accessor = device.getOneDRegisterAccessor<int>("/DAQ/MACRO_PULSE_NUMBER", 1, 0);
27
28 // Perform a full read cycle through the public API
29 accessor.read();
30
31 // After the read cycle, we have valid data
32 BOOST_CHECK(accessor.dataValidity() != ChimeraTK::DataValidity::faulty);
33}

35// ------------------------------------------------------------
36// Test that concurrent reads on the same double-buffered register
37// complete successfully. The transfer lock ensures sequential
38// access to the handshake registers, so no data corruption occurs.
39BOOST_AUTO_TEST_CASE(test_concurrent_reads_same_register) {
40 Device device;
41 device.open("(dummy?map=simpleJsonFile.jmap)");
42
43 // Two accessors on the same register
44 auto accessor1 = device.getOneDRegisterAccessor<int>("/DAQ/MACRO_PULSE_NUMBER", 1, 0);
45 auto accessor2 = device.getOneDRegisterAccessor<int>("/DAQ/MACRO_PULSE_NUMBER", 1, 0);
46
47 // Read concurrently from two threads. Both must complete without
48 // deadlock or data corruption.
49 auto future = std::async(std::launch::async, [&] { accessor1.read(); });
50 accessor2.read();
51
52 auto status = future.wait_for(std::chrono::milliseconds(200));
53 BOOST_CHECK(status == std::future_status::ready);
54}

97// ------------------------------------------------------------
98// Helper that simulates the firmware finishing a buffer and raising the
99// interrupt. An interrupt-triggered read must return that freshly finished
100// buffer (the inactive one at the time of the read).
101class AsyncDoubleBufferFixture {
102 public:
103 AsyncDoubleBufferFixture()
104 : dummy(openDeviceAndGetDummy(device)),
105 accessor(device.getOneDRegisterAccessor<uint32_t>("/DAQ/ASYNC_DBLBUF", 1, 0, {AccessMode::wait_for_new_data})),
106 enable(dummy.get(), "DAQ/DOUBLE_BUF", "ENA"), inactive(dummy.get(), "DAQ/DOUBLE_BUF", "INACTIVE_BUF_ID"),
107 buffer0(dummy.get(), "DAQ/ASYNC_DBLBUF", "BUF0"), buffer1(dummy.get(), "DAQ/ASYNC_DBLBUF", "BUF1") {
108 // Enable double buffering for index 2.
109 enable[2] = 1;
110 }
111
112 // Simulate the firmware finishing a buffer: fill the freshly finished buffer
113 // and point INACTIVE_BUF_ID at it, then raise the interrupt.
114 void firmwareFinishesBuffer(uint32_t value, uint32_t newInactiveBuffer) {
115 if(newInactiveBuffer == 1) {
116 buffer0 = value;
117 }
118 else {
119 buffer1 = value;
120 }
121 inactive[2] = newInactiveBuffer;
122 dummy->triggerInterrupt(1);
123 }
124
125 Device device;
126 boost::shared_ptr<DummyBackend> dummy;
127 OneDRegisterAccessor<uint32_t> accessor;
128 DummyRegisterAccessor<uint32_t> enable;
129 DummyRegisterAccessor<uint32_t> inactive;
130 DummyRegisterAccessor<uint32_t> buffer0;
131 DummyRegisterAccessor<uint32_t> buffer1;
132
133 private:
134 static boost::shared_ptr<DummyBackend> openDeviceAndGetDummy(Device& dev) {
135 dev.open("(dummy?map=simpleJsonFile.jmap)");
136 auto backend = boost::dynamic_pointer_cast<DummyBackend>(dev.getBackend());
137 if(!backend) {
138 BOOST_FAIL("Device did not produce a DummyBackend");
139 }
140 return backend;
141 }
142};

144// ------------------------------------------------------------
145// Test that an interrupt-triggered wait_for_new_data read of the full
146// double-buffered register returns the buffer the firmware just finished, over
147// several consecutive buffer swaps.
148BOOST_AUTO_TEST_CASE(TestInterruptDrivenReadReturnsFinishedBuffer) {
149 AsyncDoubleBufferFixture f;
150 f.device.activateAsyncRead();
151
152 // An initial value is delivered when the async domain is activated.
153 BOOST_CHECK(f.accessor.readNonBlocking());
154 BOOST_CHECK(f.accessor.dataValidity() != ChimeraTK::DataValidity::faulty);
155
156 // No further data before the firmware completes the next buffer.
157 BOOST_CHECK(!f.accessor.readNonBlocking());
158
159 // Swap 1: firmware finishes buffer0 (=100) and flips to buffer1.
160 f.firmwareFinishesBuffer(100, 1);
161 BOOST_CHECK(f.accessor.readNonBlocking());
162 BOOST_CHECK_EQUAL(f.accessor[0], 100);
163 BOOST_CHECK(!f.accessor.readNonBlocking());
164
165 // Swap 2: firmware finishes buffer1 (=200) and flips to buffer0.
166 BOOST_CHECK(!f.accessor.readNonBlocking());
167 f.firmwareFinishesBuffer(200, 0);
168 BOOST_CHECK(f.accessor.readNonBlocking());
169 BOOST_CHECK_EQUAL(f.accessor[0], 200);
170 BOOST_CHECK(!f.accessor.readNonBlocking());
171
172 // Swap 3: firmware finishes buffer0 (=300) and flips to buffer1.
173 BOOST_CHECK(!f.accessor.readNonBlocking());
174 f.firmwareFinishesBuffer(300, 1);
175 BOOST_CHECK(f.accessor.readNonBlocking());
176 BOOST_CHECK_EQUAL(f.accessor[0], 300);
177 BOOST_CHECK(!f.accessor.readNonBlocking());
178
179 // Swap 4: firmware finishes buffer1 (=400) and flips to buffer0.
180 BOOST_CHECK(!f.accessor.readNonBlocking());
181 f.firmwareFinishesBuffer(400, 0);
182 BOOST_CHECK(f.accessor.readNonBlocking());
183 BOOST_CHECK_EQUAL(f.accessor[0], 400);
184 BOOST_CHECK(!f.accessor.readNonBlocking());
185
186 f.device.close();
187}
