38 static std::pair<std::string, std::string> getInterruptControllerNameAndDescription(json const& descriptor) {
39 if(!descriptor.is_object() || descriptor.size() != 1) {
40 throw ChimeraTK::logic_error(
41 "An interrupt controller descriptor must be an object with exactly one controller key.");
42 }
43 auto controllerType = descriptor.begin().key();
44 auto controllerDescription = descriptor.front().dump();
45 return {controllerType, controllerDescription};
46 }

57 std::pair<std::string, std::string> MuxedInterruptDistributorFactory::
58 getInterruptControllerNameAndDescriptionFromCatalogue(
59 std::vector<size_t> const& subdomainID, DeviceBackend& backend) {
60 for(auto const& metaDataEntry : backend.getMetadataCatalogue()) {
61 auto const& key = metaDataEntry.first;
62 if(key[0] == '!') {
63 try {
64 auto jkey = nlohmann::json::parse(std::string({++key.begin(), key.end()})); // key without the leading !
65 auto interruptId = jkey.get<std::vector<size_t>>();
66
67 if(interruptId == subdomainID) {
68 auto jdescriptor = nlohmann::json::parse(metaDataEntry.second);
69 return getInterruptControllerNameAndDescription(jdescriptor);
70 }
71 }
72 catch(json::parse_error& e) {
73 throw ChimeraTK::logic_error("Json parse error in \"" + key + "\": " + e.what());
74 }
75 }
76 }
77 std::string subdomainIdString;
78 for(auto i : subdomainID) {
79 subdomainIdString += std::to_string(i) + ":";
80 }
81 subdomainIdString.pop_back();
82
83 throw ChimeraTK::logic_error(
84 "No interrupt controller description for SubDomain " + subdomainIdString + " in MetadataCatalogue");
85 }

89 boost::shared_ptr<MuxedInterruptDistributor> MuxedInterruptDistributorFactory::createMuxedInterruptDistributor(
90 boost::shared_ptr<SubDomain<std::nullptr_t>> parent) {
91 std::string name, description;
92 std::tie(name, description) =
93 getInterruptControllerNameAndDescriptionFromCatalogue(parent->getId(), *parent->getBackend());
94 auto recordIter = _controllerTypes.find(name);
95 if(recordIter == _controllerTypes.end()) {
96 throw ChimeraTK::logic_error("Unknown interrupt controller type \"" + name + "\"");
97 }
98 return recordIter->second.creator(description, std::move(parent));
99 }

103 void MuxedInterruptDistributorFactory::augmentCatalogue(std::vector<size_t> const& interruptId,
104 std::string const& controllerType, std::string const& controllerDescription,
105 NumericAddressedRegisterCatalogue& catalogue) {
106 // An unknown controller type and a type without an augmenter are skipped, so this is a no-op for the built-in
107 // "dummy" type and for any type that is not registered. An unknown type stays an error when a distributor is
108 // created.
109 auto recordIter = _controllerTypes.find(controllerType);
110 if(recordIter == _controllerTypes.end()) {
111 return;
112 }
113 if(!recordIter->second.augmenter) {
114 return;
115 }
116 recordIter->second.augmenter(interruptId, controllerDescription, catalogue);
117 }

125 template<typename DistributorType>
126 boost::shared_ptr<AsyncAccessorManager> MuxedInterruptDistributor::getAccessorManager(
127 std::vector<size_t> const& qualififedSubDomainId) {
128 // the qualified SubDomainId is relative. We need an absolute (fully qualified ID) in case we have to crate a SubDomain
129 assert(!qualififedSubDomainId.empty());
130 auto fullyQualifiedId = _id;
131 fullyQualifiedId.push_back(qualififedSubDomainId.front());
132
133 // we can't use try_emplace because the map contains weak pointers
134 boost::shared_ptr<SubDomain<std::nullptr_t>> subDomain;
135 auto subDomainIter = _subDomains.find(qualififedSubDomainId.front());
136 if(subDomainIter != _subDomains.end()) {
137 subDomain = subDomainIter->second.lock();
138 }
139 if(!subDomain) {
140 subDomain =
141 boost::make_shared<SubDomain<std::nullptr_t>>(_backend, fullyQualifiedId, shared_from_this(), _asyncDomain);
142 _subDomains[qualififedSubDomainId.front()] = subDomain;
143 if(_asyncDomain->unsafeGetIsActive()) {
144 // Creating a new version here is correct. Nothing has been distributed to any accessor connected to this
145 // sub-interrupt yet because we just created this subDomain.
146 activateSubDomain(*subDomain, {});
147 }
148 }
149 return subDomain->getAccessorManager<DistributorType>(qualififedSubDomainId);
150 }
