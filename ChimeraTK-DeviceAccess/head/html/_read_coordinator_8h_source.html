202 template<typename UserType>
203 inline void ReadCoordinator::Group<UserType>::rawCopy(const std::vector<uint8_t>& rawBuffer) {
204 for(const auto& consumer : consumers) {
205 if(consumer.nSamples == 0) {
206 continue;
207 }
208 auto* cooked = consumer.staging.data();
209 const auto* raw = rawBuffer.data() + consumer.byteOffset;
210 // A raw-mode group holds consumers whose UserType is the register's raw data type: an integer of the matching
211 // size and signedness (signed or unsigned). The nested if constexpr is a compile-time gate: the memcpy below is
212 // instantiated only for such types and never for any other UserType (which would be undefined behaviour). The
213 // outer integral guard comes first because std::make_unsigned_t is defined only for integral and enum types, so
214 // the unsigned counterpart may be formed only behind it; the remaining non-integral UserTypes are excluded by it.
215 // isRawType covers unsigned raw types only, while a raw-mode UserType may be signed, hence the unsigned
216 // counterpart is checked. Such a use cannot be formed through the library, so no runtime fallback exists.
217 if constexpr(std::is_integral_v<UserType>) {
218 if constexpr(ChimeraTK::isRawType<std::make_unsigned_t<UserType>>) {
219 for(size_t i = 0; i < consumer.nSamples; ++i) {
220 std::memcpy(cooked + i, raw + i * consumer.stride, sizeof(UserType));
221 }
222 }
223 }
224 }
225 }
